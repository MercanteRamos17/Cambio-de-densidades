import tkinter as tk import math class SimuladorEstabilidad(tk.Tk): def __init__(self): super().__init__() # Mantenemos el título original de la pestaña self.title("Simulador taller 1 estabilidad") self.configure(bg="#0b1120") self.geometry("1100x800") self.minsize(900, 700) # --- Variables con los valores por defecto intactos --- self.var_eslora = tk.DoubleVar(value=33) self.var_manga = tk.DoubleVar(value=7) self.var_htri = tk.DoubleVar(value=3) self.var_desp1 = tk.DoubleVar(value=1302.2625) self.var_rho1 = tk.DoubleVar(value=1.025) self.var_deltaT = tk.DoubleVar(value=0.3) self.var_rho2 = tk.DoubleVar(value=1.000) self.out_Ax = tk.StringVar(value="0.00 m²") self.out_Tsw = tk.StringVar(value="0.00 m") self.out_Tfw = tk.StringVar(value="0.00 m") self.out_w = tk.StringVar(value="0.00 t") self.crear_interfaz() self.calcular() # Rastrear cambios en tiempo real (equivalente al 'input' event listener en JS) variables = [self.var_eslora, self.var_manga, self.var_htri, self.var_desp1, self.var_rho1, self.var_deltaT, self.var_rho2] for var in variables: var.trace_add("write", lambda *args: self.calcular()) def resetear(self): self.var_eslora.set(33) self.var_manga.set(7) self.var_htri.set(3) self.var_desp1.set(1302.2625) self.var_rho1.set(1.025) self.var_deltaT.set(0.3) self.var_rho2.set(1.000) self.calcular() def crear_interfaz(self): # --- Título y Encabezado intactos --- tk.Label(self, text="EJERCICIO 4 - CAMBIO DE DENSIDADES - EMBARCACIÓN PRISMÁTICA", bg="#0b1120", fg="#38bdf8", font=("Segoe UI", 16, "bold")).pack(pady=(20, 5)) tk.Label(self, text="Docente - Carlos Velasquez Lombana - ENAP", bg="#0b1120", fg="#94a3b8", font=("Segoe UI", 12)).pack(pady=(0, 20)) # Botón de reset btn_reset = tk.Button(self, text="⟳ Restablecer Valores por Defecto", bg="#1e293b", fg="#38bdf8", activebackground="#38bdf8", activeforeground="#0b1120", relief="flat", font=("Segoe UI", 11, "bold"), command=self.resetear, cursor="hand2") btn_reset.pack(pady=(0, 25)) # Dashboard Principal main_frame = tk.Frame(self, bg="#0b1120") main_frame.pack(fill="both", expand=True, padx=40, pady=(0, 20)) # --- PANEL IZQUIERDO (Controles) --- panel_controls = tk.Frame(main_frame, bg="#1e293b", padx=20, pady=20, bd=1, relief="ridge") panel_controls.pack(side="left", fill="y", padx=(0, 20)) def crear_input(parent, label, unit, variable): f = tk.Frame(parent, bg="#1e293b") f.pack(fill="x", pady=6) tk.Label(f, text=label, bg="#1e293b", fg="#f8fafc", font=("Segoe UI", 10, "bold")).pack(side="left") tk.Label(f, text=unit, bg="#1e293b", fg="#f8fafc").pack(side="right") tk.Entry(f, textvariable=variable, width=10, justify="right", font=("Segoe UI", 10, "bold"), bg="#0b1120", fg="#f8fafc", insertbackground="#38bdf8").pack(side="right", padx=5) tk.Label(panel_controls, text="Parámetros Geométricos", bg="#1e293b", fg="#38bdf8", font=("Segoe UI", 12, "bold")).pack(anchor="w", pady=(0, 10)) crear_input(panel_controls, "Eslora (L)", "m", self.var_eslora) crear_input(panel_controls, "Manga (B)", "m", self.var_manga) crear_input(panel_controls, "Altura Triángulo (h_tri)", "m", self.var_htri) tk.Label(panel_controls, text="Condición Inicial (Mar)", bg="#1e293b", fg="#38bdf8", font=("Segoe UI", 12, "bold")).pack(anchor="w", pady=(20, 10)) crear_input(panel_controls, "Desplazamiento 1 (Δ₁)", "t", self.var_desp1) crear_input(panel_controls, "Densidad Agua Mar (ρ_SW)", "t/m³", self.var_rho1) tk.Label(panel_controls, text="Condición Final (Dulce)", bg="#1e293b", fg="#38bdf8", font=("Segoe UI", 12, "bold")).pack(anchor="w", pady=(20, 10)) crear_input(panel_controls, "Variación Calado (δT)", "m", self.var_deltaT) crear_input(panel_controls, "Densidad Agua Dulce (ρ_FW)", "t/m³", self.var_rho2) # --- PANEL DERECHO (Resultados y Visores) --- panel_der = tk.Frame(main_frame, bg="#0b1120") panel_der.pack(side="left", fill="both", expand=True) # Tarjetas de Resultados res_frame = tk.Frame(panel_der, bg="#1e293b", padx=20, pady=20, bd=1, relief="ridge") res_frame.pack(fill="x", pady=(0, 20)) tk.Label(res_frame, text="Reporte de Estabilidad", bg="#1e293b", fg="#38bdf8", font=("Segoe UI", 12, "bold")).pack(anchor="w", pady=(0, 15)) cards_frame = tk.Frame(res_frame, bg="#1e293b") cards_frame.pack(fill="x") def crear_card(parent, titulo, var, color, bg_color="#000000"): f = tk.Frame(parent, bg=bg_color, padx=10, pady=10) f.pack(side="left", expand=True, fill="both", padx=5) tk.Label(f, text=titulo, bg=bg_color, fg="#94a3b8", font=("Segoe UI", 9)).pack(anchor="w") tk.Label(f, textvariable=var, bg=bg_color, fg=color, font=("Segoe UI", 14, "bold")).pack(anchor="w", pady=(5,0)) # Simular el border-left de CSS borde = tk.Frame(f, bg=color, width=4) borde.place(relx=0, rely=0, relheight=1) # Mismos colores y estilos que tu CSS crear_card(cards_frame, "Área Max (A_X)", self.out_Ax, "#f8fafc", "#171f2e") crear_card(cards_frame, "Calado Mar (T_SW)", self.out_Tsw, "#f59e0b", "#171f2e") crear_card(cards_frame, "Calado Dulce (T_FW)", self.out_Tfw, "#10b981", "#171f2e") crear_card(cards_frame, "Provisiones (w)", self.out_w, "#ef4444", "#2c1c24") # Fondo rojizo simulado # Visores Gráficos (Canvas) canv_frame = tk.Frame(panel_der, bg="#0b1120") canv_frame.pack(fill="both", expand=True) # 2D box2d = tk.Frame(canv_frame, bg="#020617", bd=1, relief="solid") box2d.pack(side="left", fill="both", expand=True, padx=(0, 10)) tk.Label(box2d, text="VISTA TRANSVERSAL 2D", bg="#020617", fg="#38bdf8", font=("Segoe UI", 9, "bold")).pack(pady=10) self.c2d = tk.Canvas(box2d, width=400, height=300, bg="#020617", highlightthickness=0) self.c2d.pack(fill="both", expand=True, padx=10, pady=10) # 3D box3d = tk.Frame(canv_frame, bg="#020617", bd=1, relief="solid") box3d.pack(side="right", fill="both", expand=True, padx=(10, 0)) tk.Label(box3d, text="VISTA LONGITUDINAL 3D", bg="#020617", fg="#38bdf8", font=("Segoe UI", 9, "bold")).pack(pady=10) self.c3d = tk.Canvas(box3d, width=400, height=300, bg="#020617", highlightthickness=0) self.c3d.pack(fill="both", expand=True, padx=10, pady=10) def calcular(self): try: # Obtener valores L = float(self.var_eslora.get() or 1) B = float(self.var_manga.get() or 1) h_tri = float(self.var_htri.get() or 1) desp1 = float(self.var_desp1.get() or 0) rho1 = float(self.var_rho1.get() or 1) deltaT = float(self.var_deltaT.get() or 0) rho2 = float(self.var_rho2.get() or 1) except ValueError: return # Evita errores si el usuario deja el campo vacío mientras escribe # --- Lógica Matemática Intacta --- V1 = desp1 / rho1 Ax1 = V1 / L area_tri_max = (B * h_tri) / 2 if Ax1 > area_tri_max: T_sw = h_tri + ((Ax1 - area_tri_max) / B) else: T_sw = math.sqrt((2 * Ax1 * h_tri) / B) T_fw = T_sw + deltaT if T_fw > h_tri: Ax2 = area_tri_max + (B * (T_fw - h_tri)) else: Ax2 = (math.pow(T_fw, 2) * B) / (2 * h_tri) V2 = Ax2 * L desp2 = V2 * rho2 w = desp2 - desp1 # Actualizar Reporte self.out_Ax.set(f"{Ax1:.2f} m²") self.out_Tsw.set(f"{T_sw:.2f} m") self.out_Tfw.set(f"{T_fw:.2f} m") self.out_w.set(f"{w:.4f} t") max_H = max(T_sw, T_fw) + (h_tri * 0.5) # Redibujar Canvases self.c2d.update() # Asegurar dimensiones actuales self.c3d.update() w_2d, h_2d = self.c2d.winfo_width(), self.c2d.winfo_height() w_3d, h_3d = self.c3d.winfo_width(), self.c3d.winfo_height() if w_2d > 10: # Validar que ya se renderizó la ventana self.dibujar2D(B, h_tri, max_H, T_sw, T_fw, w_2d, h_2d) self.dibujar3D(L, B, h_tri, max_H, T_sw, T_fw, w_3d, h_3d) def dibujar_cota(self, ctx, x, y1, y2, texto, color, align_right=False): ctx.create_line(x, y1, x, y2, fill=color, width=2) ctx.create_line(x-5, y1, x+5, y1, fill=color, width=2) ctx.create_line(x-5, y2, x+5, y2, fill=color, width=2) anchor = "e" if align_right else "w" tx = x - 10 if align_right else x + 10 ctx.create_text(tx, y2 + (y1 - y2) / 2, text=texto, fill=color, font=("Segoe UI", 10, "bold"), anchor=anchor) def dibujar2D(self, B, h_tri, H_total, T1, T2, w, h): self.c2d.delete("all") scale = min(w / (B + 6), h / (H_total + 2)) cx, cy = w / 2, h - 30 # Malla (Grid) for i in range(0, w, 20): self.c2d.create_line(i, 0, i, h, fill="#1e293b") for i in range(0, h, 20): self.c2d.create_line(0, i, w, i, fill="#1e293b") # Agua Dulce y Salada self.c2d.create_rectangle(0, cy - (T2 * scale), w, cy, fill="#064e3b", outline="") self.c2d.create_rectangle(0, cy - (T1 * scale), w, cy, fill="#0c4a6e", outline="") # Casco puntos = [ cx, cy, cx + (B/2 * scale), cy - (h_tri * scale), cx + (B/2 * scale), cy - (H_total * scale), cx - (B/2 * scale), cy - (H_total * scale), cx - (B/2 * scale), cy - (h_tri * scale) ] self.c2d.create_polygon(puntos, fill="#0f172a", outline="#f8fafc", width=3) # Cotas self.dibujar_cota(self.c2d, cx + (B/2 * scale) + 20, cy, cy - (T1 * scale), f"{T1:.2f}m (T_SW)", "#f59e0b") self.dibujar_cota(self.c2d, cx - (B/2 * scale) - 20, cy, cy - (T2 * scale), f"{T2:.2f}m (T_FW)", "#10b981", True) def dibujar3D(self, L, B, h_tri, H_total, T1, T2, w, h): self.c3d.delete("all") anguloISO = 25 * math.pi / 180 reduccionZ = 0.4 maxX = B + L * reduccionZ * math.cos(anguloISO) maxY = H_total + L * reduccionZ * math.sin(anguloISO) scale = min((w - 100) / maxX, (h - 80) / maxY) if maxX > 0 and maxY > 0 else 1 dx = w / 2 - (B * scale / 2) - (L * reduccionZ * math.cos(anguloISO) * scale / 2) + (B * scale / 2) dy = h - 30 def proy(x, y, z): return (dx + (x * scale) + (z * scale * reduccionZ * math.cos(anguloISO)), dy - (y * scale) - (z * scale * reduccionZ * math.sin(anguloISO))) p = { "f_k": proy(0, 0, 0), "f_p": proy(-B/2, h_tri, 0), "f_s": proy(B/2, h_tri, 0), "f_dp": proy(-B/2, H_total, 0), "f_ds": proy(B/2, H_total, 0), "b_k": proy(0, 0, L), "b_p": proy(-B/2, h_tri, L), "b_s": proy(B/2, h_tri, L), "b_dp": proy(-B/2, H_total, L), "b_ds": proy(B/2, H_total, L) } def trazar(pts, fill, outline, width=1): flat_pts = [] for pt in pts: flat_pts.extend(pt) self.c3d.create_polygon(flat_pts, fill=fill, outline=outline, width=width) # Espejo (Popa) trazar([p["b_k"], p["b_s"], p["b_ds"], p["b_dp"], p["b_p"]], "#1e293b", "#94a3b8") # Planos de flotación def trazar_agua(T, color): mangaEnT = (T * B) / h_tri if T <= h_tri else B trazar([proy(-mangaEnT/2, T, 0), proy(mangaEnT/2, T, 0), proy(mangaEnT/2, T, L), proy(-mangaEnT/2, T, L)], color, "#cbd5e1") trazar_agua(T1, "#0c4a6e") # Mar if T2 > T1: trazar_agua(T2, "#064e3b") # Dulce # Aristas longitudinales self.c3d.create_line(*p["f_k"], *p["b_k"], fill="#f8fafc", width=2) self.c3d.create_line(*p["f_p"], *p["b_p"], fill="#f8fafc", width=2) self.c3d.create_line(*p["f_s"], *p["b_s"], fill="#f8fafc", width=2) self.c3d.create_line(*p["f_dp"], *p["b_dp"], fill="#f8fafc", width=2) self.c3d.create_line(*p["f_ds"], *p["b_ds"], fill="#f8fafc", width=2) # Cara Frontal (Proa) trazar([p["f_k"], p["f_s"], p["f_ds"], p["f_dp"], p["f_p"]], "#0f172a", "#f8fafc", 2) # Textos 3D p1 = proy(B/2, T1, 0) self.c3d.create_text(p1[0]+8, p1[1]+5, text=f"◀ T_SW: {T1:.2f}m", fill="#f59e0b", font=("Segoe UI", 10, "bold"), anchor="w") p2 = proy(-B/2, T2, 0) self.c3d.create_text(p2[0]-8, p2[1]+5, text=f"{T2:.2f}m :T_FW ▶", fill="#10b981", font=("Segoe UI", 10, "bold"), anchor="e") if __name__ == "__main__": app = SimuladorEstabilidad() app.mainloop()
